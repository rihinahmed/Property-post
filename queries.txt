CREATE TABLE users (
     id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
     name VARCHAR2(100) NOT NULL,
     email VARCHAR2(100) UNIQUE NOT NULL,
     password_hash VARCHAR2(255) NOT NULL,
     role VARCHAR2(20) NOT NULL
   );

CREATE TABLE rooms (
  id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id NUMBER NOT NULL,
  title VARCHAR2(100),
  description VARCHAR2(500),
  listed_date DATE DEFAULT SYSDATE,
  CONSTRAINT fk_rooms_user FOREIGN KEY (user_id) REFERENCES users(id)
);

//Blog post table from user
CREATE TABLE blog_posts (
  id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id NUMBER NOT NULL,
  title VARCHAR2(200),
  content CLOB,
  created_at DATE DEFAULT SYSDATE,
  CONSTRAINT fk_blogposts_user FOREIGN KEY (user_id) REFERENCES users(id)
);

-- Messages table linked to users
CREATE TABLE messages (
  id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id NUMBER NOT NULL,
  message_text CLOB,
  sent_at DATE DEFAULT SYSDATE,
  CONSTRAINT fk_messages_user FOREIGN KEY (user_id) REFERENCES users(id)
);

-- Announcement table

CREATE TABLE announcements (
  id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title VARCHAR2(255) NOT NULL,
  content CLOB NOT NULL,
  posted_by VARCHAR2(100),
  posted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Room table
CREATE TABLE rooms (
  room_id NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  title VARCHAR2(100),
  location VARCHAR2(100),
  rent NUMBER,
  description CLOB,
  latitude NUMBER,
  longitude NUMBER,
  amenities CLOB,
  pros CLOB,
  cons CLOB,
  images CLOB,
  created_at DATE DEFAULT SYSDATE
);

-- Creating seller table which will be generated from users table
CREATE TABLE SELLERS (
  SELLER_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  USER_ID NUMBER UNIQUE NOT NULL,
  SHOP_NAME VARCHAR2(100),
  PHONE VARCHAR2(20),
  ADDRESS VARCHAR2(255),
  CREATED_AT DATE DEFAULT SYSDATE,
  CONSTRAINT FK_SELLERS_USER FOREIGN KEY (USER_ID)
    REFERENCES USERS(ID) ON DELETE CASCADE
);

-- Property table
CREATE TABLE properties (
  id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  seller_id NUMBER NOT NULL,
  property_id VARCHAR2(50) DEFAULT SYS_GUID() UNIQUE,
  title VARCHAR2(255),
  location VARCHAR2(255),
  status VARCHAR2(50),
  views NUMBER DEFAULT 0,
  saved_count NUMBER DEFAULT 0,
  price NUMBER,
  image_url VARCHAR2(500),
  created_at DATE DEFAULT SYSDATE,
  updated_at DATE DEFAULT SYSDATE,
  CONSTRAINT fk_properties_seller FOREIGN KEY (seller_id) REFERENCES users(id)
);

BEGIN
    EXECUTE IMMEDIATE 'DROP TABLE messages';
EXCEPTION
    WHEN OTHERS THEN
        IF SQLCODE != -942 THEN
            RAISE;
        END IF;
END;
/


BEGIN
    EXECUTE IMMEDIATE 'DROP SEQUENCE messages_seq';
EXCEPTION
    WHEN OTHERS THEN
        IF SQLCODE != -2289 THEN -- ORA-02289: sequence does not exist
            RAISE;
        END IF;
END;
/

CREATE SEQUENCE messages_seq START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;


CREATE TABLE messages (
    ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    SENDER_ID NUMBER NOT NULL,
    RECIPIENT_ID NUMBER NOT NULL,
    MESSAGE_TEXT CLOB NOT NULL,
    SENT_AT TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT fk_sender FOREIGN KEY (SENDER_ID) REFERENCES users(ID),
    CONSTRAINT fk_recipient FOREIGN KEY (RECIPIENT_ID) REFERENCES users(ID)
);


COMMENT ON TABLE messages IS 'Stores private chat messages between users.';
COMMENT ON COLUMN messages.ID IS 'Unique identifier for each message.';
COMMENT ON COLUMN messages.SENDER_ID IS 'ID of the user who sent the message.';
COMMENT ON COLUMN messages.RECIPIENT_ID IS 'ID of the user who received the message.';
COMMENT ON COLUMN messages.MESSAGE_TEXT IS 'Content of the message.';
COMMENT ON COLUMN messages.SENT_AT IS 'Timestamp when the message was sent.';


CREATE INDEX idx_message_sender_recipient ON messages (SENDER_ID, RECIPIENT_ID);
